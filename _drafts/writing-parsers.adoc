== Simple tokenizer

Since we are only naive here and not crazy, so we don't even bother with
{regex}[regular expressions] and plug some conditionals and a loop together:

[source,rust]
----
const STMNT: &str = "SELECT * FROM spotify;";

fn main() {
    let mut start: isize = -1;
    let mut stop: isize = -1;

    let mut verb = "";
    let mut column = "";
    let mut table = "";
    let mut remainder = String::new();

    for (i, c) in STMNT.chars().enumerate() {
        match c {
            'A'..='Z' | 'a'..='z' => { // <.>
                if -1 == start {
                    start = i as isize;
                }

                stop = i as isize + 1;
            },
            ' ' | ';' => { // <.>
                if stop > start {
                    #[cfg(feature = "stats")]
                    println!("Token `{:?}` at {}-{}",
                        &STMNT[start as usize..stop as usize], start, stop);

                    if verb.is_empty() { // <.>
                        verb = &STMNT[start as usize..stop as usize];
                    } else {
                        table = &STMNT[start as usize..stop as usize];
                    }

                    start = -1;
                    stop = -1;
                }

                #[cfg(feature = "stats")]
                if ';' == c {
                    println!("End-of-Stmnt at {}", i);
                }
            },
            '*' => { // <.>
                #[cfg(feature = "stats")]
                println!("Wildcard `*` at {}", i);

                column = "*";
            },
            _ => remainder.push(c), // <.>
        }
    }

    println!("Parsed: verb={:?}, column={:?}, table={:?}", verb, column, table);
    println!("Parsed: remainder={:?}", remainder);
}

----
<.> Still naive, we just handle different characters here for our token handling
<.> Theses two characters have a special meaning for end of token resp. end or statement
<.> Here is a good point to actually copy the values
<.> Let us handle wildcards differently; could probably also be just allowed as token character
<.> Just for completeness, let us collect everything we didn't catch

== Tokenizer with FSM

I guess you've read this far just to see another {fsm}[FSM] and hell I don't
want to disappoint anyone here.
So let us solve this again, this this time with a bit of abstraction and
FSM-flavor and hopefully make my former university prof happy:

[source,rust]
----
use std::fmt;

const STMNT: &str = "SELECT * FROM spotify;";

#[derive(Debug)] // <.>
enum State {
    START,
    SCAN,
    TOKEN(usize, usize),
    WILDCARD(usize),
    BLANK(usize),
    #[allow(non_camel_case_types)]
    END_OF_STMNT(usize),
    END,
}

impl fmt::Display for State { // <.>
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            State::START => write!(f, "START"),
            State::SCAN => write!(f, "SCAN"),
            State::TOKEN(start, stop) => write!(f, "TOKEN({}, {})", start, stop),
            State::WILDCARD(pos) => write!(f, "WILDCARD({})", pos),
            State::BLANK(pos) => write!(f, "BLANK({})", pos),
            State::END_OF_STMNT(pos) => write!(f, "END_OF_STMNT({})", pos),
            State::END => write!(f, "END"),
        }

    }
}

fn main() {
    let mut states: Vec<State> = vec![State::START]; // <.>

    let mut iter = STMNT.chars().enumerate();
    let (mut i, mut c) = iter.next().unwrap_or((usize::MAX, char::MAX));

    let mut verb = "";
    let mut column = "";
    let mut table = "";
    let mut remainder = String::new();

    'parser: loop { // <.>
        match states.last().unwrap_or(&State::START) {
            State::START | State::SCAN => { // <.>
                match c {
                    'A'..='Z' | 'a'..='z' => {
                        states.push(State::TOKEN(i, i + 1));
                    }
                    ' ' => { // <.>
                        states.push(State::BLANK(i));
                    }
                    ';' => {
                        states.push(State::END_OF_STMNT(i));
                    }
                    '*' => {
                        states.push(State::WILDCARD(i));
                    }
                    _ => remainder.push(c),
                }
            },
            State::TOKEN(start, _) => { // <.>
                match c {
                    'A'..='Z' | 'a'..='z' => {
                        let newstate = State::TOKEN(*start, i + 1);

                        states.pop(); // <.>
                        states.push(newstate);

                        (i, c) = iter.next().unwrap_or((usize::MAX, char::MAX)); // <.>
                    }
                    _ => states.push(State::SCAN),
                }
            },
            State::WILDCARD(_) => { // <.>
                states.push(State::SCAN);
                (i, c) = iter.next().unwrap_or((usize::MAX, char::MAX));
            },
            State::BLANK(_) => {
                states.push(State::SCAN);
                (i, c) = iter.next().unwrap_or((usize::MAX, char::MAX));
            },
            State::END_OF_STMNT(_) => {
                states.push(State::END);
            },
            State::END => {
                break 'parser; // <.>
            }
        }
    }

    #[cfg(feature = "debug")] // <.>
    for state in states.iter() {
        println!("{}", state);
    }

    for state in states.iter() { // <.>
        match state {
            State::TOKEN(start, stop) => {
                #[cfg(feature = "stats")]
                println!("Token `{}` at {}-{}", &STMNT[*start..*stop], start, stop);

                if verb.is_empty() { // <.>
                    verb = &STMNT[*start..*stop];
                } else {
                    table = &STMNT[*start..*stop];
                }
            },
            #[allow(unused)] // <.>
            State::WILDCARD(pos) => {
                #[cfg(feature = "stats")]
                println!("Wildcard `*`at {}", pos);
                column = "*";
            },
            #[allow(unused)]
            State::END_OF_STMNT(pos) => {
                #[cfg(feature = "stats")]
                println!("End-of-Stmnt at {}", pos);
            },
            _ => {},
        }
    }

    println!("Parsed: verb={:?}, column={:?}, table={:?}", verb, column, table);
    println!("Parsed: remainder={:?}", remainder);
}

----
<.> We start with the definition of our parsing states and use enum variants
<.> Deriving from {debug}[Debug] is nice, but let us also roll our own formatter
<.> We keep our states neatly collected in a vec, so we can do some post-processing later
<.> Technically not required here, but adding labels to loops make it a bit easier to follow
<.> Utilizing the FSM idea basically means we  step through the various states and have some checks,
if something is reasonable within its boundaries
<.> As before, the blank is some special case and we handle it accordingly
<.> The token state is something special, because it also keeps tracks of the start and stop position
<.> Handling a vector of {option}[options] is sometimes a pain to modify, so we just pop/push to basically
compact the many states this creates
<.> Iterator are also strictly safe-guarded, so we iterate here manually whenever we have to and
avoid peeking or stepping back
<.> The other states are a bit of repetitive, I left them in to make it easier to follow though
<.> Here is the break with the label, this is easier to follow, right?
<.> Another nifty feature is the {features}[feature] handling of Cargo, this allows us to see the
state list when enabled
<.> Since we basically decoupled handling and parsing, we still need to strep through our states and
assign variables
<.> Not the best way to do that through..
<.> And the rest is just boring default stuff

== Parser combinator

When I started spotql I did a quick check how one would write idiomatic parsers
urc
with Rust and I quickly discovered the create {nom}[nom] - a parser combinators
library.
It took me quite a while until it made click and the migration from version 6
to version 8, which basically deprecated every macro in favor of functions,
didn't make it easier either.

{parsercombinator}[Parser combinators] or in general combinatory parsing works
slightly different, than how we did that in the two examples above.
So far we've created a fancy loop, which accepts an input string and after some
processing returns some kind of structure with the result.

TIP: The first parser didn't exactly do that, but technically the second did with
its states vector - consider it as a preliminary form of {ast}[AST].

Parser combinators, on the other hand, accept one or more parsers as an input
and generally return a new parser as its output - this are so called
{highorder}[higher-order functions].
I know this might sound a bit strange, but let us start from bottom and learn
about this on our way.
Keep in mind we just use plain Rust here, so nothing beyond {std}[std].

=== Carving out

Since we need functions, a good start is to move some of the actual parsing
into a function, which makes re-use dramatically easier and more DRY-ish.

We had to parse lots of things like our SQL verb or other identifiers, which is
not rocket science, but looks quite nice and easy like this:

[source,rust]
----
fn identifier(input: &str) -> Result<(&str, String), &str> { // <.>
    let mut stop: usize = 0;
    let mut iter = input.chars();

    match iter.next() { // <.>
        Some(next) if next.is_alphabetic() => stop += 1,
        _ => return Err(input),
    }

    while let Some(next) = iter.next() { // <.>
        if next.is_alphanumeric() {
            stop += 1;
        } else {
            break;
        }
    }

    Ok((&input[stop..], String::from(&input[..stop]))) // <.>
}
----
<.> The return type looks odd I know, but let us ignore it for now
<.> Identifiers basically always start with a letter
<.> Followed by more of same including numbers this time
<.> Unfortunately we have to deal with the return type now: This is basically
a result, containing a tuple of 1) the remaining input we didn't consume and
2) the actual string we've consumed and another string as error, we still ignore

=== Higher-order functions

Next up on our list are strings, which need to be there to keep the format, but
we actually have no direct interest in, so we can just gobble them up:

[source,rust]
----
fn tag(expected: &'static str) -> impl Fn(&str) -> Result<(&str, String), &str> { // <.>
    move |input| match input.get(0..expected.len()) { // <.>
        Some(next) if next == expected => { // <.>
            Ok((&input[expected.len()..], expected.to_string()))
        },
        _ => Err(input),
    }
}
----
<.> Here they are, higher-order functions with improved weirdness: This time we
return a function (marked by the {impl}[impl] keyword, which returns our result
from before.
<.> I don't want to throw {movesems}[move semantics] into the mix, just pretend
this is the weird syntax of Rust to define that
<.> And with our dutiful {guard}[guard] we filter for our wanted string and
slice out remaining and matched tag - this is not directly gobbled up, but if
you prefer it replace it with {unit}[unit].

=== Adding a combinator

Now it is getting interesting:
We have two functions now, but need a way to combine them or otherwise we'd
basically re-create the first parser, just more complex.

Take a seat and meet `pair`, our first actual combinator:

[source,rust]
----
pub(crate) fn pair<P1, P2, R1, R2>(parser1: P1, parser2: P2) -> impl Fn(&str) -> Result<(&str, (R1, R2)), &str>
where
    P1: Fn(&str) -> Result<(&str, R1), &str>,
    P2: Fn(&str) -> Result<(&str, R2), &str>, // <.>
{
    move |input| match parser1(input) {
        Ok((next_input, result1)) => match parser2(next_input) { // <.>
            Ok((final_input, result2)) => Ok((final_input, (result1, result2))),
            Err(err) => Err(err),
        },
        Err(err) => Err(err),
    }
}
----
<.> What the..? I know, I know. The syntax looks a bit oddfootnote:[Funny thing is
we are trying to parse SQL and actually find something like a `where` clause in
our function signature, how likely is that?], but this is just Rust's way of
describing {generics}[generics] in detail. `P1` and respectively `P2` must
return a function with our known result from before
<.> And on the inside we just concat both and fail early, if possible

=== Interim step

`Pair` works exactly as the named implies:
It allows to combine two parsers and returns the result when called.
This is probably easer to understand with a short example:

[source,rust]
----
fn main() {
    let parser = pair(identifier, tag("from")); // <.>

    let result = parser("SELECT from"); // <.>

    println!("{:?}", result); // <.>
}
----
<.> Throw both into a pair and receive a shiny new parser
<.> This actually calls our parser with this string

fn whitespace(input: &str) -> Result<(&str, String), &str> {

=== More mathematics

Before we can really fire this off, we need one additional element - a
{functor}[functor].
A functor is basically something, that can map between things and in our case
just call a function on our results:

[source,rust]
----
pub(crate) fn map<P, F, A, B>(parser: P, map_fn: F) -> impl Fn(&str) -> Result<(&str, B), &str>
where
    P: Fn(&str) -> Result<(&str, A), &str>,
    F: Fn(A) -> B,
{
    move |input| match parser(input) {
        Ok((next_input, result)) => Ok((next_input, map_fn(result))), // <.>
        Err(err) => Err(err),
    }
}
----
<.> We covered lots of ground today and barely start to sweat when seeing this:
We just call the passed function and return the result

=== Mapping all the things

Now with our functor, we can quickly add some more trivial helper functions `left`
and `right`:

[source,rust]
----
pub(crate) fn left<'a, P1, P2, R1, R2>(parser1: P1, parser2: P2) -> impl Fn(&str) -> Result<(&str, R1), &str>
where
    P1: Fn(&str) -> Result<(&str, R1), &str>,
    P2: Fn(&str) -> Result<(&str, R2), &str>,
{
    map(pair(parser1, parser2), |(left, _right)| left) // <.>
}

pub(crate) fn right<'a, P1, P2, R1, R2>(parser1: P1, parser2: P2) -> impl Fn(&str) -> Result<(&str, R2), &str>
where
    P1: Fn(&str) -> Result<(&str, R1), &str>,
    P2: Fn(&str) -> Result<(&str, R2), &str>,
{
    map(pair(parser1, parser2), |(_left, right)| right) // <.>
}
----
<.> Every building block in place and this is just piece of cake:
Call both parsers from the pair and return just the result of the left one.
<.> And opposite direction - call both and return right

=== Handle delimited strings

Before adding more stuff, we should focus again on what we actually want to
achieve:

Getting rid of leading and trailing whitespaces in a functional and cool way.

Since we have everything in place, let us a create a new parser called
`delimited`, that can use three parsers and like `left` and `right`, return
only a selection - here we go for the middle this time:

[source,rust]
----
pub(crate) fn delimited<'a, P1, P2, P3, R1, R2, R3>(parser1: P1, parser2: P2, parser3: P3) -> impl Fn(&str) -> Result<(&str, R2), &str>
where
    P1: Fn(&str) -> Result<(&str, R1), &str>,
    P2: Fn(&str) -> Result<(&str, R2), &str>,
    P3: Fn(&str) -> Result<(&str, R3), &str>,
{
    map(right(parser1, left(parser2, parser3)), |middle| middle) // <.>
}
----
<.> We use a combination of `right` and `left` to sort this out and call `map`
on the result to get to the middle

}

So we need something that handles either optional or zero to many.
Optional is something for another day, so we go with the last approach:

[source,rust]
----
pub(crate) fn zero_or_more<'a, P, A>(parser: P) -> impl Fn(&str) -> Result<(&str, Vec<A>), &str>
where
    P: Fn(&str) -> Result<(&str, A), &str>,
{
    move |mut input| {
        let mut result = Vec::new();

        while let Ok((next_input, next_item)) = parser(input) { // <.>
            input = next_input;
            result.push(next_item);
        }

        Ok((input, result))
    }
}
----
<.> Another loop, storing everything in a vector and never bail out - next!

=== Perfection

Took us a while, but we have all elements in place and can successfully
assemble our parser and ultimately parse the string:

[source,rust]
----
fn main() {
    let verb_parser = delimited(zero_or_more(whitespace), identifier, zero_or_more(whitespace));
    let column_parser = delimited(zero_or_more(whitespace), tag("*"), zero_or_more(whitespace));
    let table_parser = right(
        delimited(zero_or_more(whitespace), tag("FROM"), zero_or_more(whitespace)),
        left(identifier, tag(";")));

    let parser = pair(verb_parser, pair(column_parser, table_parser));

    if let Ok(remainder) = map(parser, |(verb, (column, table))| {
        println!("Parsed: verb={:?}, column={:?}, table={:?}", verb, column, table);
    })(STMNT) {
        println!("Parsed: remainder={:?}", remainder.1);
    }
}
----

proper types and a trait, to avoid repeat yourself, but this is more on the
ergonomics side:

[source,rust]
----
pub(crate) type ParseResult<'a, Output> = Result<(&'a str, Output), &'a str>;

pub(crate) trait Parser<'a, Output> {
    fn parse(&self, input: &'a str) -> ParseResult<'a, Output>;
}
----

examples, if you want to see how this can be done.
