== Example time

The wire protocol itself is quite lengthy and involves all the fun of network
protocols and {endianess}[endianess], but to keep the focus and to get
side-tracked too much, let us just pick a really simple statement:

[source,sql]
----
SELECT * FROM spotify;
----

This should be fairly easy to parse, so here we go.

== Simple tokenizer

Since we are only naive here and not batshit crazy, we don't even bother with
{regex}[regular expressions] and plug some conditionals and a loop together:

[source,rust]
----
const STMNT: &str = "SELECT * FROM spotify;";

fn main() {
    let mut start: isize = -1;
    let mut stop: isize = -1;

    for (i, c) in STMNT.chars().enumerate() {
        match c {
            'A'..='Z' | 'a'..='z' => { // <.>
                if -1 == start {
                    start = i as isize;
                }

                stop = i as isize + 1;
            },
            ' ' | ';' => { // <.>
                if stop > start {
                    println!("Token `{:?}` at {}-{}",
                        &STMNT[start as usize..stop as usize], start, stop);

                    start = -1;
                    stop = -1;
                }

                if ';' == c {
                    println!("EoS at {}", i);
                }
            },
            '*' => { // <.>
                println!("Wildcard `*` at {}", i);
            },
            _ => panic!("Huh?"),
        }
    }
}
----

== Tokenizer with FSM

I guess you've waiting for them and I cannot publish this without adding a bit
of FSM flavor to the mix.
Building upon our previous example, our next more university-tolerated
approach looks like this:

[source,rust]
----
use std::fmt;

const STMNT: &str = "SELECT * FROM spotify;";

#[derive(Debug)]
enum Tokens {
    START,
    TOKEN(usize, usize),
    WILDCARD(usize),
    BLANK(usize),
    EOS(usize),
    END,
}

impl fmt::Display for Tokens {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            Tokens::START => write!(f, "START"),
            Tokens::TOKEN(start, stop) => write!(f, "TOKEN({}, {})", start, stop),
            Tokens::WILDCARD(pos) => write!(f, "WILDCARD({})", pos),
            Tokens::BLANK(pos) => write!(f, "BLANK({})", pos),
            Tokens::EOS(pos) => write!(f, "EOS({})", pos),
            Tokens::END => write!(f, "END"),
        }

    }
}

fn main() {
    let mut states: Vec<Tokens> = vec![Tokens::START];

    for (i, c) in STMNT.chars().enumerate() {
        println!("{} => {}:{}", states.last(), i, c);

        match states.last().unwrap_or(&Tokens::START) {
            Tokens::START => {
                match c {
                    'A'..='Z' | 'a'..='z' => { // <.>
                        states.push(Tokens::TOKEN(i, i + 1));
                    },
                    _ => panic!("Unexpected char `{:?}`", c),
                }
            },
            Tokens::TOKEN(start, _) => {
                match c {
                    'A'..='Z' | 'a'..='z' => { // <.>
                        states.pop();
                        states.push(Tokens::TOKEN(*start, i + 1));
                    },
                    ' ' => {
                        states.push(Tokens::BLANK(i));
                    } ';' => { // <.>
                        states.push(Tokens::EOS(i));
                    },
                    '*' => {
                        states.push(Tokens::WILDCARD(i));
                    }
                    _ => panic!("Unexpected char `{:?}`", c),
                }
            },
            Tokens::WILDCARD(_) => {
                states.push(Tokens::START);
            }
            Tokens::BLANK(_) => {
                states.push(Tokens::START);
            },
            Tokens::EOS(_) => {
                states.push(Tokens::END);
            },
            Tokens::END => {
                // Nop
            }
        }
    }

    for state in states.iter() {
        println!("{}", state);
    }
}
----
